% N+1-bit multiplication diagram - NMSB version (N bits are MSB)
\begin{tikzpicture}[
    scale=1.45,
    transform shape,
    every node/.style={align=center,font=\small}
]

% First operand A (left rectangle)
% N bits MSB on left, 1 bit LSB on right
\draw[thick, fill=blue!20] (0,0) rectangle (3,0.4);
\draw[thick, fill=blue!20] (3,0) rectangle (4,0.4);
\node (an) at (1.5,0.2) {n};
\node (a1) at (3.5,0.2) {1};

% Coordinate nodes at bottom center of A rectangles
\coordinate (anbottom) at (1.5,0);
\coordinate (a1bottom) at (3.5,0);

% Second operand B (right rectangle)
% N bits MSB on left, 1 bit LSB on right
\draw[thick, fill=green!20] (5,0) rectangle (8,0.4);
\draw[thick, fill=green!20] (8,0) rectangle (9,0.4);
\node (bn) at (6.5,0.2) {n};
\node (b1) at (8.5,0.2) {1};

% Coordinate nodes at bottom center of B rectangles
\coordinate (bnbottom) at (6.5,0);
\coordinate (b1bottom) at (8.5,0);

% Computational graph nodes
\node[circle, draw, thick, minimum size=0.15cm, fill=orange!30] (mulnn) at (1.5,-1.0) {$\times$};
\node[circle, draw, thick, minimum size=0.15cm, fill=purple!20] (andn1) at (4,-1.0) {\&};
\node[circle, draw, thick, minimum size=0.15cm, fill=purple!20] (andn2) at (6.5,-1.0) {\&};
\node[circle, draw, thick, minimum size=0.15cm, fill=purple!20] (and1) at (9,-1.0) {\&};

% Connections from operands to computational nodes
% A_N and B_N to Mul NxN
\draw[->, thick] (anbottom) -- (mulnn);
\draw[->, thick] (bnbottom) -- (mulnn);

% B_N and A_1 to AndN (andn1)
\draw[->, thick] (bnbottom) -- (andn1);
\draw[->, thick] (a1bottom) -- (andn1);

% A_N and B_1 to AndN (andn2)
\draw[->, thick] (anbottom) -- (andn2);
\draw[->, thick] (b1bottom) -- (andn2);

% A_1 and B_1 to And1
\draw[->, thick] (a1bottom) -- (and1);
\draw[->, thick] (b1bottom) -- (and1);

% Addition node for combining two AndN outputs
\node[circle, draw, thick, minimum size=0.1cm, fill=cyan!30, below=-0.30cm of andn1, xshift=1.25cm] (add) {$+$};

\begin{scope}[yshift=0.25cm]
% Adder Tree bounding box - centered and spanning the width of both rectangles
\node[draw, thick, fill=teal!10, minimum width=9.5cm, minimum height=1.8cm] (addertree) at (4.5,-3.3) {};
\node[anchor=north west] at (0,-2.5) {Adder Tree};

\begin{scope}[yshift=-0.10cm]

% Inside Adder Tree: three stacked rectangles showing shift and data portions (right-aligned)
% All rectangles end at x=8, widths proportional to bit counts
% P1: 2N+2 bits, P2+P3: N+2 bits, P4: 1 bit
% Using proportional widths relative to maximum (2N+2)

% P1: 1 bit, no shift
\draw[thick, fill=white] (7.82,-2.45) rectangle (8,-2.85);
\node at (7.91,-2.65) {\footnotesize 1};
\node[anchor=west] at (8.2,-2.65) {\small P1};

% P2+P3 (AddN output): 2N+3 bits total with N+1-bit yellow extension on left
\draw[thick, fill=yellow!30] (5,-3.0) rectangle (6.42,-3.4); % N+1-bit yellow extension on left (MSB)
\draw[thick, fill=white] (6.42,-3.0) rectangle (7.82,-3.4); % Data portion (N+1 bits)
\draw[thick, fill=gray!40] (7.82,-3.0) rectangle (8,-3.4); % Shift portion (1 bit) on right - same width as P1
\node at (5.71,-3.2) {\small n+1};
\node at (7.12,-3.2) {\small n};
\node at (7.91,-3.2) {\footnotesize 1};
\node[anchor=west] at (8.2,-3.2) {\small P2};

% Red dotted rectangle enclosing 1 LSB from P1 and P2+P3
\draw[blue!80!black, line width=1.2pt, densely dotted] (7.78,-2.4) rectangle (8.05,-3.45);

% P4: 2N+2 bits total, 2N bits data (white on left), 2 bits shift (gray on right) - WIDEST
% Divide 2N data to align with P2+P3: into N + N, plus 1 bit, plus 1 bit sections
\draw[thick, fill=white] (5,-3.55) rectangle (7.72,-3.95); % Data portion (2N bits)
\draw[thick, fill=gray!40] (7.72,-3.55) rectangle (8,-3.95); % Shift portion (2 bits) on right
% Subdivide 2N into sections that align with N+1
\draw[thick] (6.36,-3.55) -- (6.36,-3.95); % Divider: N | N
\draw[thick] (7.86,-3.55) -- (7.86,-3.95); % Divider in shift portion: 1 | 1 (aligned with P1 and P2+P3)
\node at (5.68,-3.75) {\small n};
\node at (7.04,-3.75) {\small n};
\node at (7.79,-3.75) {\footnotesize 1};
\node at (7.93,-3.75) {\footnotesize 1};
\node[anchor=west] at (8.2,-3.75) {\small P3};

% Red dotted rectangle enclosing 2 LSB bits from P2+P3 and 2 LSB bits from P4
\draw[red!85!black, line width=1.2pt, densely dotted] (7.70,-2.95) rectangle (8.05,-4.0);
\end{scope}
\end{scope}

% Connections from computational nodes to addition node
\draw[->, thick] (andn1) -- (add);
\draw[->, thick] (andn2) -- (add);

% Connections directly to Adder Tree
\draw[->, thick] (mulnn) -- (mulnn |- addertree.north) node[near start, right] {P3};
\draw[->, thick] (add) -- (add |- addertree.north) node[near start, right, yshift=-2pt] {P2};
\draw[->, thick] (and1) -- (and1 |- addertree.north) node[near start, right] {P1};

\end{tikzpicture}
